USE PetShopManagementDB;
GO

-- ১. Users টেবিল
CREATE TABLE Users (
    UserId INT PRIMARY KEY IDENTITY(1,1),
    Username VARCHAR(50) UNIQUE NOT NULL,
    Password VARCHAR(255) NOT NULL,
    FullName VARCHAR(100),
    UserRole VARCHAR(20) CHECK (UserRole IN ('Admin', 'Customer', 'DeliveryMan')) NOT NULL,
    Email VARCHAR(100),
    Phone VARCHAR(20),
    Address TEXT,
    CreatedAt DATETIME DEFAULT GETDATE()
);
GO

-- ২. Categories টেবিল
CREATE TABLE Categories (
    CategoryId INT PRIMARY KEY IDENTITY(1,1),
    CategoryName VARCHAR(50) NOT NULL,
    Description TEXT
);
GO

-- ৩. Products টেবিল
CREATE TABLE Products (
    ProductId INT PRIMARY KEY IDENTITY(1,1),
    ProductName VARCHAR(100) NOT NULL,
    CategoryId INT FOREIGN KEY REFERENCES Categories(CategoryId),
    Price DECIMAL(18, 2) NOT NULL,
    StockQuantity INT NOT NULL,
    ProductType VARCHAR(50),
    ImageUrl VARCHAR(MAX)
);
GO

-- ৪. Coupons টেবিল
CREATE TABLE Coupons (
    CouponId INT PRIMARY KEY IDENTITY(1,1),
    CouponCode VARCHAR(20) UNIQUE NOT NULL,
    DiscountPercentage DECIMAL(5, 2) NOT NULL,
    ExpiryDate DATE,
    IsActive BIT DEFAULT 1
);
GO

-- ৫. Orders টেবিল
CREATE TABLE Orders (
    OrderId INT PRIMARY KEY IDENTITY(1,1),
    CustomerId INT FOREIGN KEY REFERENCES Users(UserId),
    DeliveryManId INT FOREIGN KEY REFERENCES Users(UserId) NULL,
    OrderDate DATETIME DEFAULT GETDATE(),
    TotalAmount DECIMAL(18, 2),
    CouponId INT FOREIGN KEY REFERENCES Coupons(CouponId) NULL,
    OrderStatus VARCHAR(20) DEFAULT 'Pending',
    CashMemoNo AS ('INV-' + CAST(OrderId AS VARCHAR(10)))
);
GO

-- ৬. Inventory টেবিল
CREATE TABLE Inventory (
    ID INT PRIMARY KEY IDENTITY(1,1),
    Category NVARCHAR(50) NOT NULL,
    Breed NVARCHAR(50) NOT NULL,
    Price DECIMAL(18, 2) NOT NULL,
    Quantity INT NOT NULL,
    SellCount INT DEFAULT 0
);
GO

-- ৭. Sales টেবিল
CREATE TABLE Sales (
    SaleId INT PRIMARY KEY IDENTITY(1,1),
    CustomerUsername VARCHAR(50) NOT NULL,
    ProductName VARCHAR(100) NOT NULL,
    Price DECIMAL(10, 2) NOT NULL,
    Quantity INT DEFAULT 1,
    SaleDate DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_UserSales FOREIGN KEY (CustomerUsername) REFERENCES Users(Username)
);
GO

-- ৮. OrderDetails টেবিল
CREATE TABLE OrderDetails (
    OrderDetailId INT IDENTITY(1,1) PRIMARY KEY,
    OrderId INT FOREIGN KEY REFERENCES Orders(OrderId),
    ProductName NVARCHAR(100),
    Price DECIMAL(18, 2),
    Quantity INT
);
GO

-- ৯. ProductReviews টেবিল
CREATE TABLE ProductReviews (
    ReviewId INT IDENTITY(1,1) PRIMARY KEY,
    Username VARCHAR(50) NOT NULL,
    ItemName NVARCHAR(50) NOT NULL,
    Comment NVARCHAR(MAX) NOT NULL,
    Rating INT DEFAULT 5 CHECK (Rating >= 1 AND Rating <= 5),
    ReviewDate DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_ReviewUser FOREIGN KEY(Username) REFERENCES Users(Username)
);
GO

-- ১০. DeliveryPartners টেবিল
CREATE TABLE DeliveryPartners (
    AgentID INT PRIMARY KEY IDENTITY(1,1),
    AgentName VARCHAR(100) UNIQUE,
    BusinessEarned DECIMAL(18, 2) DEFAULT 0.00,
    DeliveryCount INT DEFAULT 0
);
GO
